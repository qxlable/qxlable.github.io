def digital_root(n):
    n_str = str(n)#将整数转化为字符串
    totle = sum(int(char) for char in n_str) #强制类型转换
    while totle // 10 != 0:
        t_str = str(totle)
        totle = sum(int(char) for char in t_str)
    return totle

def digital_root(n):
    return n if n < 10 else digital_root(sum(map(int,str(n))))
